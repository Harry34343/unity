using UnityEngine;
using UnityEngine.EventSystems;

public class ObtenerPieza : MonoBehaviour, IDropHandler
{
    public void OnDrop(PointerEventData eventData)
    {
        if (eventData.pointerDrag != null && eventData.pointerDrag.CompareTag("Pieza"))
        {
            Pieza pieza = eventData.pointerDrag.GetComponent<Pieza>();
            pieza.enCuadrante = true;
            eventData.pointerDrag.transform.position = transform.position;
        }
    }
}
