using UnityEngine;
using UnityEngine.InputSystem;

public class MoverPieza : MonoBehaviour
{
    private Vector3 posicionInicial;
    private bool arrastrando = false;
    private float distancia;
    private float alturaFija;

    private static MoverPieza piezaSeleccionada;

    void OnMouseDown()
    {
        piezaSeleccionada = this;

        arrastrando = true;
        posicionInicial = transform.position;
        distancia = Vector3.Distance(transform.position, Camera.main.transform.position);
        alturaFija = transform.position.y;
    }

    void OnMouseDrag()
    {
        if (arrastrando)
        {
            Vector2 mousePos = Mouse.current.position.ReadValue();
            Ray ray = Camera.main.ScreenPointToRay(mousePos);
            Vector3 punto = ray.GetPoint(distancia);

            transform.position = new Vector3(
                punto.x,
                alturaFija,
                punto.z
            );
        }
    }

    void OnMouseUp()
    {
        if (arrastrando)
        {
            Collider[] colisiones = Physics.OverlapSphere(transform.position, 2f);

            bool enZona = false;

            foreach (Collider col in colisiones)
            {
                if (col.CompareTag("Area"))
                {
                    enZona = true;
                    break;
                }
            }

            if (!enZona)
            {
                transform.position = posicionInicial;
            }
        }

        arrastrando = false;
    }

    void Update()
    {
        if (piezaSeleccionada == this)
        {
            if (Keyboard.current.leftArrowKey.wasPressedThisFrame)
            {
                transform.Rotate(0, -90f, 0);
            }

            if (Keyboard.current.rightArrowKey.wasPressedThisFrame)
            {
                transform.Rotate(0, 90f, 0);
            }
        }
    }
}
